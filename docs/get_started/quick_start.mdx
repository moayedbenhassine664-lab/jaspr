def make_image(path, label):
    img = Image.new("RGB", (800, 500), color=(45, 55, 72))  # slate color
    draw = ImageDraw.Draw(img)
    try:
        # Try to use a common font
        font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 48)
    except Exception:
        font = ImageFont.load_default()
    w, h = draw.textsize(label, font=font)
    draw.text(((800-w)/2, (500-h)/2), label, font=font, fill=(255,255,255))
    img.save(path, "PNG")

class CalisthenicsApp extends StatefulWidget {

class _CalisthenicsAppState extends State<CalisthenicsApp> {
  bool darkMode = false;
  String locale = 'en'; // 'en' or 'ar'

  @override
  void initState() {
    super.initState();
    darkMode = widget.prefs.getBool('darkMode') ?? false;
    locale = widget.prefs.getString('locale') ?? 'en';
  }

  void toggleTheme(bool v) {
    setState(() {
      darkMode = v;
      widget.prefs.setBool('darkMode', darkMode);
    });
  }

  void toggleLocale(String lang) {
    setState(() {
      locale = lang;
      widget.prefs.setString('locale', locale);
    });
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Calisthenics Master',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        brightness: darkMode ? Brightness.dark : Brightness.light,
        primarySwatch: Colors.indigo,
      ),
      home: HomePage(prefs: widget.prefs, locale: locale, onToggleTheme: toggleTheme, onToggleLocale: toggleLocale),
    );
  }
}

class HomePage extends StatefulWidget {

class DetailPage extends StatefulWidget {
  final MoveModel move;
  final SharedPreferences prefs;
  final String locale;
  DetailPage({required this.move, required this.prefs, required this.locale});

  @override
  _DetailPageState createState() => _DetailPageState();
}

class _DetailPageState extends State<DetailPage> {
  double progress = 0.0;
  TextEditingController notesController = TextEditingController();

  @override
  void initState() {
    super.initState();
    progress = widget.prefs.getDouble('${widget.move.id}_progress') ?? 0.0;
    notesController.text = widget.prefs.getString('${widget.move.id}_notes') ?? '';
  }

  Future<void> saveAll() async {
    await widget.prefs.setDouble('${widget.move.id}_progress', progress);
    await widget.prefs.setString('${widget.move.id}_notes', notesController.text);
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(widget.locale=='ar' ? 'تم الحفظ' : 'Saved')));
  }

  @override
  Widget build(BuildContext context) {
    final isAr = widget.locale == 'ar';
    return Scaffold(
      appBar: AppBar(title: Text(widget.move.name[isAr ? 'ar' : 'en']!)),
      body: Padding(
        padding: EdgeInsets.all(14),
        child: ListView(
          children: [
            Image.asset(widget.move.asset, height: 180, fit: BoxFit.contain),
            SizedBox(height: 12),
            Text(widget.move.description[isAr ? 'ar' : 'en']!),
            SizedBox(height: 12),
            Text(isAr ? 'التقدم' : 'Progress'),
            SizedBox(height: 6),
            Row(
              children: [
                IconButton(onPressed: () => setState(() => progress = (progress-0.1).clamp(0.0,1.0)), icon: Icon(Icons.remove)),
                Expanded(child: LinearProgressIndicator(value: progress)),
                IconButton(onPressed: () => setState(() => progress = (progress+0.1).clamp(0.0,1.0)), icon: Icon(Icons.add)),
              ],
            ),
            SizedBox(height: 10),
            ElevatedButton.icon(onPressed: saveAll, icon: Icon(Icons.save), label: Text(isAr ? 'حفظ' : 'Save')),
            SizedBox(height: 16),
            Text(isAr ? 'ملاحظاتك' : 'Your notes'),
            SizedBox(height: 6),
            TextField(controller: notesController, maxLines: 6, decoration: InputDecoration(border: OutlineInputBorder())),
            SizedBox(height: 10),
            ElevatedButton(onPressed: saveAll, child: Text(isAr ? 'حفظ الملاحظات' : 'Save notes')),
          ],
        ),
      ),
    );
  }
}

class SettingsPage extends StatefulWidget {
  final SharedPreferences prefs;

  final Function(bool) onToggleTheme;

  _SettingsPageState createState() => _SettingsPageState();

class _SettingsPageState extends State<SettingsPage> {
  bool darkMode = false;
  String locale = 'en';

  @override
  void initState() {
    super.initState();
    darkMode = widget.prefs.getBool('darkMode') ?? false;
    locale = widget.prefs.getString('locale') ?? widget.locale;

  void save() {
    widget.prefs.setBool('darkMode', darkMode);
    widget.prefs.setString('locale', locale);
    widget.onToggleTheme(darkMode);
    widget.onToggleLocale(locale);
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(locale=='ar' ? 'تم الحفظ' : 'Saved')));
  }

  @override
  Widget build(BuildContext context) {
    final isAr = widget.locale=='ar';
    return Scaffold(
      appBar: AppBar(title: Text(isAr ? 'الإعدادات' : 'Settings')),
      body: Padding(
        padding: EdgeInsets.all(14),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            SwitchListTile(
              title: Text(isAr ? 'الوضع الداكن' : 'Dark mode'),
              value: darkMode,
              onChanged: (v) => setState(() => darkMode = v),
            ),
            SizedBox(height: 8),
            Text(isAr ? 'اللغة' : 'Language'),
            Row(
              children: [
                ElevatedButton(onPressed: () => setState(() => locale='en'), child: Text('English')),
                SizedBox(width: 8),
                ElevatedButton(onPressed: () => setState(() => locale='ar'), child: Text('العربية')),
              ],
            ),
            SizedBox(height: 20),
            ElevatedButton(onPressed: save, child: Text(isAr ? 'حفظ' : 'Save')),
          ],
        ),
      ),
    );
  }
}
